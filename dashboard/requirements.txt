# Pinned exact (not >=): newer Streamlit releases (confirmed through 1.59.1)
# have a st.selectbox regression — clicking a searchable selectbox doesn't
# clear the input, so typing appends to the current selection instead of
# searching. 1.40.0 is confirmed bug-free via direct browser testing. Don't
# loosen this pin without re-testing selectbox click-and-type behavior.
streamlit==1.40.0
# numpy and requests are imported directly by app code but were arriving only
# as transitive deps (via pandas and streamlit). Listed explicitly so a future
# dependency change in either parent cannot silently break the deployed build.
numpy>=1.24.0
requests>=2.31.0
gspread>=6.1.2
google-auth>=2.40.2
google-auth-oauthlib>=1.2.1
pandas>=2.0.0
plotly>=5.0.0
python-dotenv>=1.0.0
google-genai>=1.0.0
# Tableau ingestion. The Mission Finding Summary view exports as .xlsx (pandas
# needs openpyxl to read it) and its monthly summary is a PDF, and both are
# uploaded through the Finding Funnel page — so they are runtime dependencies
# of the deployed app, not dev-only tools.
openpyxl>=3.1.0
pypdf>=4.0.0
# The council packet (Informes → Generar paquete) is drawn with ReportLab and
# nothing else: platypus for the page, reportlab.graphics for the charts, the
# base-14 fonts for the type. Deliberately NOT matplotlib, Kaleido or
# WeasyPrint — no packages.txt, no headless browser, no cairo/Pango, which is
# the likeliest way to break the Streamlit Cloud build. It pulls only pillow
# and charset-normalizer, both already resolved here, and it does not move the
# streamlit pin above (verified 2026-09-21).
reportlab>=4.0,<6
